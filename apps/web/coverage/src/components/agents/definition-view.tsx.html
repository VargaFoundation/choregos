function InstructionsText({ text }: { text: string }) {
  const [all, setAll] = useState(false);
  if (!text) return <p className="text-xs text-ink-muted">No instructions: the playbook of its role speaks.</p>;
  const lines = text.split("\n");
  const long = lines.length > INSTRUCTION_LINES;
  return (
    <div className="space-y-2">
      <pre aria-label="Instructions" className="whitespace-pre-wrap break-words rounded bg-surface-muted p-3 text-xs">
        {long && !all ? lines.slice(0, INSTRUCTION_LINES).join("\n") : text}
      </pre>
      {long && (
        <button type="button" className="text-xs underline" aria-expanded={all} onClick={() => setAll(!all)}>
          {all ? "Show fewer lines" : `Show all ${lines.length} lines`}
        </button>
      )}
    </div>
  );
}

/**

 * goes to that section of its settings; an organisation agent runs in any project, so it says so and

export function ProjectSetting({
  agent,
  section,
  children,
  organisation,
}: {
  agent: Pick<Agent, "scope" | "project"> | undefined;
  section: ProjectSettingsSection;
  children: ReactNode;
  organisation?: { href: string; label: string };
}) {

      </span>
    </>
  );
  if (project) {
    return (
      <span data-testid={`inherited-${section}`}>
        <Link href={`/p/${project}/settings#${section}`} className="underline">
          {children}
        </Link>
        <span className="text-ink-muted"> ({project})</span>
        {also}
      </span>
    );
  }
  return (
    <span data-testid={`inherited-${section}`}>
      {children}{" "}
      <span className="text-ink-muted">
        {"\u2014"} set in{" "}
        <Link href="/" className="underline">

    </span>
  );
